      path_deets_routes$trigger_timestamp <- Sys.time()
      save(path_deets_routes, file = paste0(path,"images/routes/",RouteID,".RData"))
  }

    if (nrow(scheduled_departures) > 0) {
      scheduled_departures$StartTime <- gsub("T"," ",scheduled_departures$StartTime)

  }
